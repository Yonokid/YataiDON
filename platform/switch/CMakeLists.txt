cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../../CMakeLists.txt" _root_project REGEX "^project\\(YataiDON VERSION")
string(REGEX MATCH "[0-9]+\\.[0-9]+\\.[0-9]+" _root_version "${_root_project}")
project(YataiDONSwitch VERSION ${_root_version} LANGUAGES C CXX)
if(NOT SWITCH AND NOT CMAKE_SYSTEM_NAME STREQUAL "NintendoSwitch")
  message(FATAL_ERROR "Use devkitPro's Switch.cmake toolchain")
endif()
set(SWITCH TRUE)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
# Apply the release level before adding dependencies; devkitPro defaults to -O2.
foreach(language C CXX)
  string(REGEX REPLACE "(^|[ \t])-O(0|1|2|3|s|z|g|fast)" "\\1"
    release_flags "${CMAKE_${language}_FLAGS_RELEASE}")
  string(STRIP "${release_flags}" release_flags)
  set(CMAKE_${language}_FLAGS_RELEASE "${release_flags} -O3" CACHE STRING
    "Switch ${language} release flags" FORCE)
endforeach()
set(YATAIDON_SHADERC "" CACHE FILEPATH "Host shaderc from bgfx v1.167.9590-598")
if(NOT EXISTS "${YATAIDON_SHADERC}")
  message(FATAL_ERROR "Set YATAIDON_SHADERC to the matching host shaderc executable")
endif()
set(game_root "${CMAKE_CURRENT_SOURCE_DIR}/../..")
set(vendor "${CMAKE_CURRENT_SOURCE_DIR}/vendor")
if(NOT EXISTS "${vendor}/bgfx.cmake/CMakeLists.txt")
  message(FATAL_ERROR "Run python3 platform/switch/prepare-game.py first")
endif()
set(SWITCH_MESA_PREFIX "$ENV{DEVKITPRO}/portlibs/switch" CACHE PATH "mesa-switch SDK installation")
include(vulkan.cmake)
find_package(PkgConfig REQUIRED)
# Only target metadata is allowed; never resolve host multimedia archives.
set(ENV{PKG_CONFIG_LIBDIR} "${SWITCH_MESA_PREFIX}/lib/pkgconfig:$ENV{DEVKITPRO}/portlibs/switch/lib/pkgconfig")
set(ENV{PKG_CONFIG_PATH} "")
pkg_check_modules(SWITCH_MEDIA REQUIRED libavformat libavcodec libavutil libswscale libswresample)
set(YATAIDON_MEDIA_LIBRARY_DIR "" CACHE PATH "Optional matching rebuilt media archives")
if(YATAIDON_MEDIA_LIBRARY_DIR AND NOT IS_DIRECTORY "${YATAIDON_MEDIA_LIBRARY_DIR}")
  message(FATAL_ERROR "YATAIDON_MEDIA_LIBRARY_DIR does not exist: ${YATAIDON_MEDIA_LIBRARY_DIR}")
endif()
set(switch_media_libraries)
foreach(library IN LISTS SWITCH_MEDIA_STATIC_LIBRARIES)
  if(YATAIDON_MEDIA_LIBRARY_DIR AND EXISTS "${YATAIDON_MEDIA_LIBRARY_DIR}/lib${library}.a")
    list(APPEND switch_media_libraries "${YATAIDON_MEDIA_LIBRARY_DIR}/lib${library}.a")
  else()
    list(APPEND switch_media_libraries "${library}")
  endif()
endforeach()

set(BGFX_BUILD_TOOLS OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BGFX_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BGFX_LIBRARY_TYPE STATIC CACHE STRING "" FORCE)
set(BGFX_CONFIG_MULTITHREADED ON CACHE BOOL "" FORCE)
set(BGFX_CONFIG_VIDEO OFF CACHE BOOL "" FORCE)
set(BGFX_CONFIG_RENDERER_VULKAN 1 CACHE STRING "" FORCE)
add_subdirectory("${vendor}/bgfx.cmake" "${CMAKE_BINARY_DIR}/bgfx" EXCLUDE_FROM_ALL)
target_compile_definitions(bx PUBLIC _DEFAULT_SOURCE PRIVATE BX_CONFIG_CRT_PROCESS=0)
target_compile_definitions(bgfx PRIVATE BGFX_CONFIG_RENDERER_VULKAN=1 BGFX_CONFIG_RENDERER_NVN=0 BGFX_CONFIG_RENDERER_OPENGLES=0 BGFX_PLATFORM_SUPPORTS_SPIRV=1)
target_link_libraries(bgfx PRIVATE Vulkan::Vulkan)
include(deps.cmake)
include(sdl.cmake)

file(GLOB_RECURSE game_sources CONFIGURE_DEPENDS "${game_root}/src/*.cpp")
list(FILTER game_sources EXCLUDE REGEX "/src/platform/|/src/libs/optional/|/gfx_raylib\\.cpp$")
add_executable(YataiDON ${game_sources} "${game_root}/src/platform/platform_switch.cpp" sqlite-vfs.c heap.c input.cpp keyboard.cpp)
target_link_options(YataiDON PRIVATE "-Wl,--wrap=__nx_exit")
target_include_directories(YataiDON PRIVATE
  ${SWITCH_MEDIA_INCLUDE_DIRS}
  "${game_root}/src" "${CMAKE_BINARY_DIR}/generated"
  "${vendor}/rapidjson/include" "${vendor}/tomlplusplus/include"
  "${vendor}/spdlog/include" "${vendor}/sol2/include"
  "${vendor}/raylib/src/external")
target_compile_definitions(YataiDON PRIVATE YATAIDON_USE_SDL2 SDL_MAIN_HANDLED
  YATAIDON_GFX_BGFX=1 YATAIDON_SNDFILE_PCM_ONLY
  $<$<CONFIG:Debug>:SOL_ALL_SAFETIES_ON=1>)
target_link_directories(YataiDON PRIVATE ${SWITCH_MEDIA_STATIC_LIBRARY_DIRS})
target_link_libraries(YataiDON PRIVATE bgfx bx yataidon_lua yataidon_sqlite yataidon_qrcodegen
  sndfile samplerate miniz ebur128 SDL2-static ${switch_media_libraries} Vulkan::Vulkan)
target_compile_options(YataiDON PRIVATE ${SWITCH_MEDIA_CFLAGS_OTHER})
target_link_options(YataiDON PRIVATE ${SWITCH_MEDIA_STATIC_LDFLAGS_OTHER})
find_package(Iconv REQUIRED)
if(YATAIDON_MEDIA_LIBRARY_DIR AND EXISTS "${YATAIDON_MEDIA_LIBRARY_DIR}/libiconv.a")
  get_target_property(iconv_target_type Iconv::Iconv TYPE)
  if(iconv_target_type STREQUAL "INTERFACE_LIBRARY")
    set_property(TARGET Iconv::Iconv PROPERTY INTERFACE_LINK_LIBRARIES "${YATAIDON_MEDIA_LIBRARY_DIR}/libiconv.a")
  else()
    set_property(TARGET Iconv::Iconv PROPERTY IMPORTED_LOCATION "${YATAIDON_MEDIA_LIBRARY_DIR}/libiconv.a")
  endif()
endif()
target_link_libraries(YataiDON PRIVATE Iconv::Iconv)
target_compile_definitions(YataiDON PRIVATE YATAIDON_HAVE_ICONV)
include(shaders.cmake)
set(font_output "${CMAKE_BINARY_DIR}/generated/default_font.cpp")
add_custom_command(OUTPUT "${font_output}"
  COMMAND ${CMAKE_COMMAND} -DIN=${game_root}/src/libs/gfx/default.ttf
    -DOUT=${font_output} -DNAME=default_font_ttf_data -P ${game_root}/cmake/embed.cmake
  DEPENDS "${game_root}/src/libs/gfx/default.ttf" "${game_root}/cmake/embed.cmake" VERBATIM)
target_sources(YataiDON PRIVATE "${font_output}")
nx_generate_nacp("${CMAKE_BINARY_DIR}/YataiDON.nacp"
  NAME "YataiDON" AUTHOR "YataiDON contributors" VERSION "${PROJECT_VERSION}")
nx_create_nro(YataiDON NACP "${CMAKE_BINARY_DIR}/YataiDON.nacp")
